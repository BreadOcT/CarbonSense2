#include <WiFi.h>
#include <Firebase_ESP_Client.h>
#include <WiFiManager.h>

#include "addons/TokenHelper.h"
#include "addons/RTDBHelper.h"

#define API_KEY "AIzaSyD421U8mdS4GT-E4ETxBkfz4A3VLgRtlFM"
#define DATABASE_URL "https://carbonsense-iot-default-rtdb.asia-southeast1.firebasedatabase.app"

FirebaseData fbdo;
FirebaseAuth auth;
FirebaseConfig config;

const int sensorPin = 34; // Pin Analog MQ-135
const int buttonPin = 0;  // Tombol BOOT bawaan ESP32 untuk Reset WiFi

unsigned long previousMillis = 0;
const long interval = 2000; // Kirim data setiap 2 detik

void setup() {
  Serial.begin(115200);
  pinMode(sensorPin, INPUT);
  pinMode(buttonPin, INPUT_PULLUP);

  Serial.println("\n[CarbonSense] Memulai sistem telemetri...");
  WiFiManager wm;
  bool res = wm.autoConnect("Setup_CarbonSense");

  if (!res) {
    Serial.println("Gagal terhubung ke WiFi. Restarting...");
    ESP.restart();
  }

  config.api_key = API_KEY;
  config.database_url = DATABASE_URL;

  Firebase.begin(&config, &auth);
  Firebase.reconnectWiFi(true);
}

void loop() {
  if (millis() - previousMillis >= interval) {
    previousMillis = millis();

    int rawValue = analogRead(sensorPin);
    // Konversi ADC ke skala PPM gas MQ-135
    float ppm = map(rawValue, 0, 4095, 300, 2000);

    Serial.printf("Kadar PPM Gas Terbaca: %.1f\n", ppm);
    Firebase.RTDB.setFloat(&fbdo, "/devices/CS-ESP32-LIVING-01/ppm", ppm);
    Firebase.RTDB.setString(&fbdo, "/devices/CS-ESP32-LIVING-01/lastActive", "Baru saja");
  }
}
